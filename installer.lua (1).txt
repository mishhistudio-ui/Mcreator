  state.path = path

local function viewFile(path)
  local data = fs.read(path) or ""
  local cut = #data > 40000
  if cut then data = data:sub(1, 40000) end
  local lines = {}
  for line in (data .. "\n"):gmatch("(.-)\r?\n") do
    lines[#lines + 1] = (line:gsub("\t", "  "))
  end
  if cut then lines[#lines] = "… файл слишком большой, показано начало" end
  state.view = { path = path, lines = lines }
  state.viewScroll = 0
  win.setTitle("Файлы — " .. path)
end

local function sizeText(bytes)
  bytes = floor(tonumber(bytes) or 0)
  if bytes < 1024 then return bytes .. " Б" end
  return string.format("%.1f КБ", bytes / 1024)
end

  local path = join(state.path, item.name)

  for i = 1, #snake do

  end

end

local function placeFood()
  repeat
    food = { math.random(1, COLS), math.random(1, ROWS) }
  until not occupied(food[1], food[2])
end

local function reset()
  snake = {}
  for i = 1, 6 do snake[i] = { 16 - i, 20 } end
  dx, dy, nextDx, nextDy = 1, 0, 1, 0
  score, alive = 0, true
  placeFood()
end

win.onTick = function()
  if not started or not alive then return false end
  dx, dy = nextDx, nextDy
  local x, y = snake[1][1] + dx, snake[1][2] + dy
  if x < 1 or x > COLS or y < 1 or y > ROWS or occupied(x, y) then
    alive = false
    if score > best then
      best = score
      app.save({ best = best })
    end
    return
  end
  table.insert(snake, 1, { x, y })
  if x == food[1] and y == food[2] then
    score = score + 1
    placeFood()
  else
    snake[#snake] = nil
  end
end

  local x2, y2 = x + w - 1, y + h - 1
  if x < cx1 then x = cx1 end
  if y < cy1 then y = cy1 end
  if x2 > cx2 then x2 = cx2 end
  if y2 > cy2 then y2 = cy2 end
  if x > x2 or y > y2 then return end
  fg = fg or 0
  ch = ch or " "
  for j = y, y2 do
    local i = (j - 1) * W + x
    for _ = x, x2 do
      newBg[i], newFg[i], newCh[i] = bg, fg, ch
      i = i + 1
    end
  end
end

-- Текст. Если bg не указан, фон ячеек сохраняется («прозрачный» текст).
function screen.text(x, y, fg, str, bg)
  if y < cy1 or y > cy2 then return end
  local i = (y - 1) * W + x
  for c in gmatch(str, CHAR) do
    if x > cx2 then break end
    if x >= cx1 then
      newCh[i], newFg[i] = c, fg
      if bg then newBg[i] = bg end
    end
    x = x + 1
    i = i + 1
  end
end

-- «Пиксель» половинной высоты: y считается в половинках строк (1..H*2).
function screen.pixel(x, y, color)
  local row = floor((y + 1) / 2)
  if x < cx1 or x > cx2 or row < cy1 or row > cy2 then return end
  local i = (row - 1) * W + x
  local top, bottom
  if newCh[i] == "▀" then
    top, bottom = newFg[i], newBg[i]
  else
    top, bottom = newBg[i], newBg[i]
  end
  if y % 2 == 1 then top = color else bottom = color end
  newCh[i], newFg[i], newBg[i] = "▀", top, bottom
end

local function dim(color)
  local cached = dimCache[color]
  if cached then return cached end
  local r, g, b = floor(color / 65536) % 256, floor(color / 256) % 256, color % 256
  cached = floor(r * 0.55) * 65536 + floor(g * 0.55) * 256 + floor(b * 0.55)
  dimCache[color] = cached
  return cached
end

-- Затемняет уже нарисованную область (используется для теней окон).

  if x < cx1 then x = cx1 end
  if y < cy1 then y = cy1 end
  if x2 > cx2 then x2 = cx2 end
  if y2 > cy2 then y2 = cy2 end

    for _ = x, x2 do
      newBg[i], newFg[i] = dim(newBg[i]), dim(newFg[i])
      i = i + 1
    end
  end

  if win.onResize then safeCall(win.app, win.onResize, win.width, win.height) end
  dirty = true
end

function system.notify(title, text)
  toasts[#toasts + 1] = { title = tostring(title), text = tostring(text or ""), expires = computer.uptime() + 4 }
  dirty = true
end

function system.alert(title, text)
  alertSeq = alertSeq + 1
  local inst = {
    info = { id = "alert:" .. alertSeq, name = tostring(title), icon_char = "!", icon_color = theme.danger },
    windows = {}, internal = true,
  }
  running[#running + 1] = inst
  local lines = wrap(text, 52)
  if #lines > 14 then
    for i = #lines, 15, -1 do lines[i] = nil end
    lines[14] = "…"
  end
  local win = newWindow(inst, { title = title, width = 56, height = #lines + 4, resizable = false })
  win.onDraw = function(g)
    for i, line in ipairs(lines) do g.text(3, i + 1, line) end
    g.button(g.width - 13, g.height - 1, 12, "Понятно", function() closeWindow(win) end, { primary = true })
  end
  return win
end

local function crash(inst, reason)
  if inst.crashed then return end
  inst.crashed = true
  local list = { unpack(inst.windows) }
  for _, win in ipairs(list) do closeWindow(win, true) end
  removeRunning(inst)
  if inst.internal then return end
  reason = tostring(reason):gsub("\nstack traceback:.*", "")
  system.alert("«" .. inst.info.name .. "» остановлено", "Приложение завершилось с ошибкой:\n" .. reason)
end

local function processCrashes()
  while #crashes > 0 do
    local item = table.remove(crashes, 1)
    crash(item[1], item[2])
  end
end

local g = { wrap = wrap, shorten = shorten }
local gx, gy, gw, gwin = 1, 1, 1, nil

function g.fill(x, y, w, h, bg, fg, ch)
  screen.rect(gx + floor(x) - 1, gy + floor(y) - 1, floor(w), floor(h), bg or theme.window, fg or theme.text, ch or " ")
end

function g.clear(bg)
  screen.rect(gx, gy, gw, g.height, bg or theme.window, theme.text, " ")
end

function g.text(x, y, text, fg, bg)
  screen.text(gx + floor(x) - 1, gy + floor(y) - 1, fg or theme.text, tostring(text), bg)
end

function g.center(y, text, fg, bg)
  text = tostring(text)
  g.text(floor((gw - unicode.len(text)) / 2) + 1, y, text, fg, bg)
end

-- «пиксель» половинной высоты: y от 1 до height * 2
function g.pixel(x, y, color)
  screen.pixel(gx + floor(x) - 1, (gy - 1) * 2 + floor(y), color)
end

function g.icon(x, y, ch, color)
  drawIcon(gx + floor(x) - 1, gy + floor(y) - 1, ch, color)
end

-- область, реагирующая на нажатие: fn(x, y, кнопка) с координатами внутри области
function g.zone(x, y, w, h, fn)
  local zones = gwin.zones
  zones[#zones + 1] = { floor(x), floor(y), floor(w), floor(h), fn }
end

function g.button(x, y, w, label, fn, opts)
  opts = opts or {}
  local h = opts.height or 1
  local bg = opts.color or (opts.primary and theme.accent or theme.field)
  local fg = opts.textColor or (opts.primary and theme.accentText or theme.text)
  if opts.disabled then bg, fg = theme.field, theme.subtext end
  label = tostring(label)
  g.fill(x, y, w, h, bg)
  g.text(x + floor((w - unicode.len(label)) / 2), y + floor((h - 1) / 2), label, fg)
  if fn and not opts.disabled then g.zone(x, y, w, h, fn) end
end

-- поле ввода: значение хранится в model[key]
function g.input(x, y, w, model, key, opts)
  opts = opts or {}
  local win = gwin
  local active = win == focused and win.focusModel == model and win.focusKey == key
  local value = tostring(model[key] or "")
  g.fill(x, y, w, 1, theme.field)
  if active then g.text(x, y, "▌", theme.accent) end
  if value == "" and not active then
    g.text(x + 1, y, shorten(opts.placeholder or "", w - 2), theme.subtext)
  else
    local shown = value
    if opts.mask then shown = string.rep("•", unicode.len(value)) end
    local room = w - 3
    local len = unicode.len(shown)
    if len > room then shown = unicode.sub(shown, len - room + 1) end
    g.text(x + 1, y, shown, theme.text)
    if active then g.text(x + 1 + unicode.len(shown), y, "▏", theme.accent) end
  end
  win.inputs[#win.inputs + 1] = { model = model, key = key, opts = opts }
  g.zone(x, y, w, 1, function()
    win.focusModel, win.focusKey, win.focusOpts = model, key, opts
  end)
end

local function makeContext(inst)
  local dataPath = "/NIOS/Data/" .. inst.info.id:gsub("[^%w%-]", "_") .. ".cfg"
  local app = {
    name = inst.info.name,
    theme = theme,
    screen = { width = W, height = H },
    keys = { up = 200, down = 208, left = 203, right = 205, enter = 28, space = 57, back = 14, tab = 15, esc = 1 },
  }
  function app.window(opts) return newWindow(inst, opts) end
  function app.redraw() dirty = true end
  function app.notify(text) system.notify(inst.info.name, text) end
  function app.save(value) return fs.writeTable(dataPath, value) end
  function app.load() return fs.readTable(dataPath) end
  function app.time() return computer.uptime() end
  function app.quit()
    local list = { unpack(inst.windows) }
    for _, win in ipairs(list) do closeWindow(win) end
  end
  if inst.info.system then app.system = system end
  return app
end

function launch(info)
  local inst = runningById(info.id)
  if inst then
    local win = inst.windows[#inst.windows]
    if win then focus(win) end
    return inst
  end
  local source = fs.read(info.path)
  if not source then
    system.alert("Не удалось запустить", "Файл приложения «" .. tostring(info.name) .. "» не найден:\n" .. tostring(info.path))
    return
  end
  local env = setmetatable({}, { __index = _G })
  local chunk, reason = load(source, "=" .. tostring(info.name), "t", env)
  if not chunk then
    system.alert("Ошибка в коде «" .. tostring(info.name) .. "»", reason)
    return
  end
  inst = { info = info, windows = {} }
  running[#running + 1] = inst
  local ok, why = xpcall(chunk, traceback, makeContext(inst))
  if not ok then
    crash(inst, why)
    return
  end
  if #inst.windows == 0 then removeRunning(inst) end
  dirty = true
  return inst
end

function system.launch(info) return launch(info) end

function system.launchById(id)
  for _, info in ipairs(apps) do
    if info.id == id then return launch(info) end
  end
end

function system.shutdown(reboot)
  computer.shutdown(reboot and true or false)
end

local function drawWallpaper()
  for y = 1, H do
    screen.rect(1, y, W, 1, wallpaper[y * 2], wallpaper[y * 2 - 1], "▀")
  end
end

local function drawDesktop()
  desktopZones = {}
  local perColumn = max(1, floor((H - TASKBAR - 2) / 5))
  local column, row = 0, 0
  for _, info in ipairs(apps) do
    if info.desktop then
      local x, y = 3 + column * 13, 2 + row * 5
      drawIcon(x + 3, y, info.icon_char, info.icon_color)
      local label = shorten(info.name, 12)
      local lx = x + floor((12 - unicode.len(label)) / 2)
      if selectedIcon == info.id then
        screen.rect(x, y + 3, 12, 1, theme.accent, theme.accentText, " ")
        screen.text(lx, y + 3, theme.accentText, label)
      else
        screen.text(lx, y + 3, textOn(wallpaper[(y + 3) * 2]), label)
      end
      desktopZones[#desktopZones + 1] = { x, y, 12, 4, info }
      row = row + 1
      if row >= perColumn then row, column = 0, column + 1 end
    end
  end
end

local function drawWindow(win)
  local active = win == focused
  local bar = active and theme.title or theme.titleInactive
  local ink = active and theme.text or theme.subtext
  if not win.maximized then
    screen.shade(win.x + win.w, win.y + 1, 2, win.h)
    screen.shade(win.x + 2, win.y + win.h, win.w - 2, 1)
  end
  screen.rect(win.x, win.y, win.w, 1, bar, ink, " ")
  screen.text(win.x + 1, win.y, floor(tonumber(win.app.info.icon_color) or theme.accent), "■")
  screen.text(win.x + 3, win.y, ink, shorten(win.title, win.w - 14))
  local bx = win.x + win.w - 3
  screen.text(bx, win.y, ink, " ✕ ")
  if win.resizable then
    bx = bx - 3
    screen.text(bx, win.y, ink, win.maximized and " ◱ " or " □ ")
  end
  screen.text(bx - 3, win.y, ink, " ─ ")

  screen.rect(win.x, win.y + 1, win.w, win.h - 1, theme.window, theme.text, " ")
  screen.setClip(win.x, win.y + 1, win.x + win.w - 1, win.y + win.h - 1)
  gx, gy, gw, gwin = win.x, win.y + 1, win.w, win
  g.width, g.height = win.w, win.h - 1
  win.zones, win.inputs = {}, {}
  if win.onDraw then safeCall(win.app, win.onDraw, g) end
  screen.resetClip()
end

local function taskbarItems()
  local items, seen = { { kind = "start" } }, {}
  for _, info in ipairs(apps) do
    if info.pinned then
      items[#items + 1] = { kind = "app", info = info }
      seen[info.id] = true
    end
  end
  for _, inst in ipairs(running) do
    if not seen[inst.info.id] then
      items[#items + 1] = { kind = "app", info = inst.info }
      seen[inst.info.id] = true
    end
  end
  return items
end

local function drawTaskbar()
  local y = H - TASKBAR + 1
  screen.rect(1, y, W, TASKBAR, theme.taskbar, theme.text, " ")
  local items = taskbarItems()
  local itemW = 6
  local x = floor((W - #items * itemW) / 2) + 1
  taskbarZones = {}
  for _, item in ipairs(items) do
    local color, ch, inst
    if item.kind == "start" then
      color, ch = theme.accent, "NI"
    else
      color = floor(tonumber(item.info.icon_color) or theme.accent)
      ch = shorten(tostring(item.info.icon_char or "◆"), 4)
      inst = runningById(item.info.id)
    end
    local current = (item.kind == "start" and startOpen)
      or (inst and focused and focused.app == inst and not startOpen)
    local plate = current and theme.card or theme.taskbar
    if current then screen.rect(x, y, itemW, TASKBAR, plate, theme.text, " ") end
    screen.rect(x + 1, y, 4, 1, plate, color, "▄")
    screen.rect(x + 1, y + 1, 4, 1, color, textOn(color), " ")
    screen.text(x + 1 + floor((4 - unicode.len(ch)) / 2), y + 1, textOn(color), ch)
    if inst then
      screen.rect(x + 2, y + 2, 2, 1, plate, current and theme.accent or theme.subtext, "▄")
    end
    taskbarZones[#taskbarZones + 1] = { x, x + itemW - 1, item }
    x = x + itemW
  end
  local time, date = clockText()
  screen.text(W - 18, y + 1, theme.subtext, date)
  screen.text(W - 6, y + 1, theme.text, time)
end

local function drawStart()
  startZones = {}
  local mw = min(58, W - 4)
  local mh = min(25, H - TASKBAR - 2)
  local mx = floor((W - mw) / 2) + 1
  local my = H - TASKBAR - mh
  startRect = { mx, my, mw, mh }
  screen.shade(mx + mw, my + 1, 2, mh)
  screen.rect(mx, my, mw, mh, theme.window, theme.text, " ")
  screen.text(mx + 3, my + 1, theme.text, "Приложения")

  local columns = max(1, floor((mw - 4) / 13))
  local visibleRows = max(1, floor((mh - 7) / 5))
  local totalRows = ceil(#apps / columns)
  startScroll = max(0, min(startScroll, totalRows - visibleRows))
  if totalRows > visibleRows then
    screen.text(mx + mw - 22, my + 1, theme.subtext, "колесо мыши — ещё ↕")
  end
  local pad = floor((mw - columns * 13) / 2)
  for i, info in ipairs(apps) do
    local index = i - 1
    local r = floor(index / columns) - startScroll
    local c = index % columns
    if r >= 0 and r < visibleRows then
      local x, y = mx + pad + c * 13, my + 3 + r * 5
      drawIcon(x + 3, y, info.icon_char, info.icon_color)
      local label = shorten(info.name, 12)
      screen.text(x + floor((12 - unicode.len(label)) / 2), y + 3, theme.text, label)
      startZones[#startZones + 1] = { x, y, 12, 4, function()
        startOpen = false
        launch(info)
      end }
    end
  end

  local fy = my + mh - 3
  screen.rect(mx, fy, mw, 3, theme.panel, theme.text, " ")
  local user = store.session and store.session.user
  screen.text(mx + 3, fy + 1, theme.accent, "●")
  screen.text(mx + 5, fy + 1, theme.text, shorten(user and user.username or "Гость", mw - 38))
  local bx = mx + mw - 30
  screen.rect(bx, fy + 1, 14, 1, theme.field, theme.text, " ")
  screen.text(bx + 1, fy + 1, theme.text, "Перезагрузка")
  startZones[#startZones + 1] = { bx, fy + 1, 14, 1, function() system.shutdown(true) end }
  screen.rect(bx + 16, fy + 1, 12, 1, theme.field, theme.text, " ")
  screen.text(bx + 17, fy + 1, theme.text, "Выключение")
  startZones[#startZones + 1] = { bx + 16, fy + 1, 12, 1, function() system.shutdown(false) end }
end

local function drawToast()
  local toast = toasts[1]
  if not toast then return end
  local w = min(40, W - 2)
  local lines = wrap(toast.text, w - 4)
  local shown = min(#lines, 3)
  local h = shown + 3
  local x, y = W - w - 1, H - TASKBAR - h
  screen.shade(x + 2, y + h, w, 1)
  screen.rect(x, y, w, h, theme.card, theme.text, " ")
  screen.rect(x, y, 1, h, theme.card, theme.accent, "▌")
  screen.text(x + 2, y + 1, theme.accent, shorten(toast.title, w - 4))
  for i = 1, shown do screen.text(x + 2, y + 1 + i, theme.text, lines[i]) end
end

function draw()
  dirty = false
  drawWallpaper()
  drawDesktop()
  for _, win in ipairs(windows) do
    if not win.minimized then drawWindow(win) end
  end
  drawTaskbar()
  if startOpen then drawStart() end
  drawToast()
  screen.flush()
end

local function inside(x, y, zx, zy, zw, zh)
  return x >= zx and x < zx + zw and y >= zy and y < zy + zh
end

local function windowAt(x, y)
  for i = #windows, 1, -1 do
    local win = windows[i]
    if not win.minimized and inside(x, y, win.x, win.y, win.w, win.h) then return win end
  end
end

local function onTaskbar(x)
  for _, zone in ipairs(taskbarZones) do
    if x >= zone[1] and x <= zone[2] then
      local item = zone[3]
      if item.kind == "start" then
        startOpen = not startOpen
        startScroll = 0
        if startOpen then pcall(store.reload) end
        return
      end
      startOpen = false
      local inst = runningById(item.info.id)
      if not inst then
        launch(item.info)
        return
      end
      local win = inst.windows[#inst.windows]
      if not win then return end
      if win.minimized or focused ~= win then
        focus(win)
      else
        win.minimized = true
        focusTop()
      end
      return
    end
  end
  startOpen = false
end

local function onTouch(x, y, button)
  dirty = true
  if startOpen then
    if inside(x, y, startRect[1], startRect[2], startRect[3], startRect[4]) then
      for _, zone in ipairs(startZones) do
        if inside(x, y, zone[1], zone[2], zone[3], zone[4]) then
          zone[5]()
          return
        end
      end
      return
    end
    if y <= H - TASKBAR then startOpen = false end
  end
  if y > H - TASKBAR then
    onTaskbar(x)
    return
  end

  local win = windowAt(x, y)
  if win then
    selectedIcon = nil
    focus(win)
    if y == win.y then
      local rx = x - win.x + 1
      if rx > win.w - 3 then
        closeWindow(win)
      elseif rx > win.w - 6 then
        if win.resizable then
          toggleMaximize(win)
        else
          win.minimized = true
          focusTop()
        end
      elseif win.resizable and rx > win.w - 9 then
        win.minimized = true
        focusTop()
      elseif not win.maximized then
        dragging = { win = win, dx = x - win.x, dy = y - win.y }
      end
      return
    end
    local cx, cy = x - win.x + 1, y - win.y
    local hit
    for i = #win.zones, 1, -1 do
      local zone = win.zones[i]
      if inside(cx, cy, zone[1], zone[2], zone[3], zone[4]) then
        hit = zone
        break
      end
    end
    win.focusModel, win.focusKey, win.focusOpts = nil, nil, nil
    captured = win
    if hit then
      safeCall(win.app, hit[5], cx - hit[1] + 1, cy - hit[2] + 1, button)
    elseif win.onTouch then
      safeCall(win.app, win.onTouch, cx, cy, button)
    end
    return
  end

  for _, zone in ipairs(desktopZones) do
    if inside(x, y, zone[1], zone[2], zone[3], zone[4]) then
      local info = zone[5]
      if selectedIcon == info.id then
        selectedIcon = nil
        launch(info)
      else
        selectedIcon = info.id
      end
      return
    end
  end
  selectedIcon = nil
end

local function onDrag(x, y, button)
  if dragging then
    local win = dragging.win
    win.x = max(4 - win.w, min(W - 3, x - dragging.dx))
    win.y = max(1, min(H - TASKBAR, y - dragging.dy))
    dirty = true
  elseif captured and captured.onDrag and not captured.closed then
    safeCall(captured.app, captured.onDrag, x - captured.x + 1, y - captured.y, button)
    dirty = true
  end
end

local function onDrop(x, y, button)
  if dragging then
    dragging = nil
  elseif captured and captured.onDrop and not captured.closed then
    safeCall(captured.app, captured.onDrop, x - captured.x + 1, y - captured.y, button)
    dirty = true
  end
  captured = nil
end

local function onScroll(x, y, direction)
  if startOpen and inside(x, y, startRect[1], startRect[2], startRect[3], startRect[4]) then
    startScroll = max(0, startScroll - direction)
    dirty = true
    return
  end
  local win = windowAt(x, y)
  if win and win.onScroll then
    safeCall(win.app, win.onScroll, x - win.x + 1, y - win.y, direction)
    dirty = true
  end
end

local function editInput(win, char, code)
  local model, key = win.focusModel, win.focusKey
  local opts = win.focusOpts or {}
  local value = tostring(model[key] or "")
  if code == 14 then
    model[key] = unicode.sub(value, 1, -2)
    if opts.onChange then safeCall(win.app, opts.onChange, model[key]) end
  elseif code == 28 then
    if opts.onEnter then safeCall(win.app, opts.onEnter, value) end
  elseif code == 15 then
    local inputs = win.inputs
    for i, entry in ipairs(inputs) do
      if entry.model == model and entry.key == key then
        local nextEntry = inputs[i % #inputs + 1]
        win.focusModel, win.focusKey, win.focusOpts = nextEntry.model, nextEntry.key, nextEntry.opts
        break
      end
    end
  elseif char >= 32 and char ~= 127 then
    if unicode.len(value) < (opts.maxLength or 200) then
      model[key] = value .. unicode.char(char)
      if opts.onChange then safeCall(win.app, opts.onChange, model[key]) end
    end
  end
end

local function onKey(char, code)
  if startOpen then
    if code == 1 then
      startOpen = false
      dirty = true
    end
    return
  end
  local win = focused
  if not win or win.minimized then return end
  dirty = true
  if win.focusModel then
    editInput(win, char, code)
  elseif win.onKey then
    safeCall(win.app, win.onKey, char, code)
  end
end

local function onKeyUp(char, code)
  local win = focused
  if win and not win.minimized and win.onKeyUp then
    safeCall(win.app, win.onKeyUp, char, code)
    dirty = true
  end
end

local function onPaste(text)
  local win = focused
  if not win or win.minimized then return end
  if win.focusModel then
    local opts = win.focusOpts or {}
    text = text:gsub("[\r\n]", " ")
    local value = tostring(win.focusModel[win.focusKey] or "") .. text
    local limit = opts.maxLength or 200
    if unicode.len(value) > limit then value = unicode.sub(value, 1, limit) end
    win.focusModel[win.focusKey] = value
    if opts.onChange then safeCall(win.app, opts.onChange, value) end
    dirty = true
  elseif win.onPaste then
    safeCall(win.app, win.onPaste, text)
    dirty = true

local function nextTimeout()

  for _, win in ipairs(windows) do

local function runTimers()
